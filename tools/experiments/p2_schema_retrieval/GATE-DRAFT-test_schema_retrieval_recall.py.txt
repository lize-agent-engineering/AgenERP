"""🔴 门禁 · schema 检索：自然语言问句 → 目标字段的 Top-5 命中率

判据来源：WBS `02-WBS.md` §5 P2.0R —— 「在本站点真实 DocType 上，自然语言问句
→ 目标字段的 **Top-5 命中率 ≥ 90%**」。

## 它防的是什么

P2.3（视图 Agent）与整个 P3（操作 Agent）都建立在 schema 检索之上。
检索不准，视图会画错字段，写入 Agent 会改错单据 —— **后者是脏账，不是错答案。**

## 它**不**验什么（写清楚，免得有人以为验了）

- **不验 oracle 上界。** 给定 DocType 时 Top-5 = 97.5%，那不是可交付的检索器。
- **不验 DocType 级召回。** 那是中间量，不是用户拿到的东西。
"""

REQUIRED_TOP5 = 90.0

# 索引范围：**business 全量**。见文件头 ③——live 范围在本站点上限只有 87.5%。
SCOPE = "business"


@pytest.mark.live
def test_natural_language_questions_find_the_right_field_in_top_five(live_site):
    """端到端：问句 → 字段，Top-5 命中率 ≥ 90%。"""
    retriever = _build_retriever(live_site)          # ← 待实现（今天仓内没有）
    items = json.loads(EVAL_SET.read_text(encoding="utf-8"))["items"]

    # ⚠️ 评测集自己先得过硬约束 ④：每条 ground truth 必须指回真实存在的字段。
    # 一份指不回真实字段的评测集，量出来的命中率没有意义。
    _assert_every_ground_truth_field_exists_on_the_site(live_site, items)

    hits = 0
    misses = []
    for item in items:
        top5 = retriever.top_fields(item["q"], k=5)
        if set(top5) & set(item["expected"]):
            hits += 1
        else:
            misses.append({"q": item["q"], "expected": item["expected"], "got": top5})

    rate = hits * 100 / len(items)
    # ⚠️ **不许写成 `assert rate >= REQUIRED_TOP5 or <任何东西>`。**
    # 一个 `or` 就是 P1 那次误放行的形状。失败时把 miss 全列出来，
    # 让红的那一次自己说清红在哪 —— 但那是 message，不是逃逸。
    assert rate >= REQUIRED_TOP5, (
        f"Top-5 命中率 {rate:.1f}% < {REQUIRED_TOP5}%（{hits}/{len(items)}）。"
        f"未命中 {len(misses)} 条：\n"
        + "\n".join(f"  · {m['q']}\n    期望 {m['expected']}\n    拿到 {m['got']}" for m in misses)
    )
